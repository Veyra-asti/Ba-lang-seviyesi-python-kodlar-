<script>
  // Menu toggle
  function toggleMenu() {
    document.getElementById('sideMenu').classList.toggle('open');
    document.getElementById('hamburger').classList.toggle('active');
  }

  // File name display
  function showFileName(input, targetId) {
    const target = document.getElementById(targetId);
    target.textContent = input.files[0] ? '✓ ' + input.files[0].name : '';
  }

  // Validate .txt only
  function validateTxt(input) {
    if (input.files[0] && !input.files[0].name.toLowerCase().endsWith('.txt')) {
      showToast('Sadece .txt dosyaları kabul edilir!', true);
      input.value = '';
      document.getElementById('scriptFileName').textContent = '';
    }
  }

  // Toast notification
  function showToast(msg, isError = false) {
    const t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.toggle('error', isError);
    t.classList.add('show');
    setTimeout(() => t.classList.remove('show'), 3500);
  }

  // Load shares from localStorage
  function loadShares() {
    const shares = JSON.parse(localStorage.getItem('astigcode_shares') || '[]');
    const container = document.getElementById('sharesContainer');
    const noShares = document.getElementById('noShares');

    if (shares.length === 0) {
      noShares.style.display = 'block';
      return;
    }

    noShares.style.display = 'none';
    container.innerHTML = '';

    shares.reverse().forEach(share => {
      const item = document.createElement('div');
      item.className = 'share-item';

      let content = '';
      if (share.type === 'bot') {
        content = `
          <div class="share-header">
            <div class="share-title">${share.name || 'Discord Bot'}</div>
            <span class="share-type bot">Bot</span>
          </div>
          <div class="share-meta">📅 ${share.date} • 👤 Anonim</div>
          <div class="share-content">
            <strong>Amacı:</strong><br>${share.purpose}<br><br>
            <strong>Komutlar:</strong><br>${share.commands}
          </div>
        `;
      } else {
        content = `
          <div class="share-header">
            <div class="share-title">${share.name}</div>
            <span class="share-type script">Skript</span>
          </div>
          <div class="share-meta">📅 ${share.date} • 👤 Anonim</div>
          <div class="share-content">
            <strong>Açıklama:</strong><br>${share.description}<br><br>
            ${share.commands ? `<strong>Komutlar:</strong><br>${share.commands}<br><br>` : ''}
            ${share.permission ? `<strong>LuckPerms İzni:</strong> ${share.permission}<br><br>` : ''}
            <strong>Kod:</strong>
            <div class="share-code">${share.code}</div>
          </div>
        `;
      }

      item.innerHTML = content;
      container.appendChild(item);
    });
  }

  // Submit Discord Bot
  function submitBot(e) {
    e.preventDefault();
    const form = e.target;
    const file = form.botFile.files[0];

    if (file && file.name.toLowerCase().endsWith('.exe')) {
      showToast('.exe dosyaları kabul edilmez!', true);
      return;
    }

    const share = {
      type: 'bot',
      name: 'Discord Bot',
      purpose: form.purpose.value,
      commands: form.commands.value,
      date: new Date().toLocaleDateString('tr-TR'),
      fileName: file ? file.name : null
    };

    const shares = JSON.parse(localStorage.getItem('astigcode_shares') || '[]');
    shares.push(share);
    localStorage.setItem('astingcode_shares', JSON.stringify(shares));

    showToast('✓ Discord bot başarıyla paylaşıldı!');
    form.reset();
    document.getElementById('botFileName').textContent = '';
    loadShares();
  }

  // Submit Script
  function submitScript(e) {
    e.preventDefault();
    const form = e.target;
    const file = form.scriptFile.files[0];

    if (file && !file.name.toLowerCase().endsWith('.txt')) {
      showToast('Sadece .txt dosyaları kabul edilir!', true);
      return;
    }

    const share = {
      type: 'script',
      name: form.name.value,
      commands: form.commands.value,
      permission: form.permission.value,
      description: form.description.value,
      code: form.code.value,
      date: new Date().toLocaleDateString('tr-TR'),
      fileName: file ? file.name : null
    };

    const shares = JSON.parse(localStorage.getItem('astigcode_shares') || '[]');
    shares.push(share);
    localStorage.setItem('astingcode_shares', JSON.stringify(shares));

    showToast('✓ Skript başarıyla paylaşıldı!');
    form.reset();
    document.getElementById('scriptFileName').textContent = '';
    loadShares();
  }

  // Smooth scroll
  document.querySelectorAll('a[href^="#"]').forEach(a => {
    a.addEventListener('click', function(e) {
      e.preventDefault();
      const target = document.querySelector(this.getAttribute('href'));
      if (target) target.scrollIntoView({ behavior: 'smooth', block: 'start' });
    });
  });

  // Close menu on outside click
  document.addEventListener('click', (e) => {
    const menu = document.getElementById('sideMenu');
    const ham = document.getElementById('hamburger');
    if (menu.classList.contains('open') && !menu.contains(e.target) && !ham.contains(e.target)) {
      toggleMenu();
    }
  });

  // Load shares on page load
  loadShares();
</script>
